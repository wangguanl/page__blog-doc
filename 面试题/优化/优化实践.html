// Intersection Observer 方式
const images = document.querySelectorAll('img[data-src]');

const imageObserver = new IntersectionObserver((entries, observer) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      const img = entry.target;
      img.src = img.dataset.src;
      img.removeAttribute('data-src');
      observer.unobserve(img);
    }
  });
});

images.forEach(img => imageObserver.observe(img));

3.2 避免回流重绘 ​

核心目标：批量修改 DOM，统一操作样式，避免频繁读取修改布局。

批量修改 DOM ​

javascript
// ❌ 不好：频繁修改，多次回流
const container = document.getElementById('list');
for (let i = 0; i < 1000; i++) {
  const div = document.createElement('div');
  div.textContent = i;
  container.appendChild(div);  // 每次都引起回流
}

// ✅ 好：使用 DocumentFragment
const container = document.getElementById('list');
const fragment = document.createDocumentFragment();

for (let i = 0; i < 1000; i++) {
  const div = document.createElement('div');
  div.textContent = i;
  fragment.appendChild(div);
}

container.appendChild(fragment);  // 只回流一次

先读后写 ​

javascript
// ❌ 不好：频繁读写混合
const element = document.getElementById('box');
element.style.left = element.offsetLeft + 10 + 'px';
element.style.top = element.offsetTop + 10 + 'px';
element.style.width = element.offsetWidth - 10 + 'px';

// ✅ 好：先读后写
const element = document.getElementById('box');
const left = element.offsetLeft;
const top = element.offsetTop;
const width = element.offsetWidth;

element.style.left = (left + 10) + 'px';
element.style.top = (top + 10) + 'px';
element.style.width = (width - 10) + 'px';

// ✅ 最好：使用 transform（不触发回流）
element.style.transform = 'translate(10px, 10px) scale(0.9)';

统一操作样式 ​

javascript
// ❌ 不好：逐个修改
element.style.color = 'red';
element.style.fontSize = '14px';
element.style.padding = '10px';

// ✅ 好：一次性修改
element.style.cssText = 'color: red; font-size: 14px; padding: 10px;';

// ✅ 更好：使用 class
element.classList.add('active');

3.3 长列表优化 ​

核心目标：大数据列表使用虚拟滚动，只渲染可视区域 DOM，杜绝节点过多导致卡顿。

使用开源库 ​

javascript
// React：react-window
import { FixedSizeList } from 'react-window';

function MyList({ items }) {
  const Row = ({ index, style }) => (
    <div style={style}>
      Item {items[index]}
    </div>
  );

  return (
    <FixedSizeList
      height={600}
      itemCount={items.length}
      itemSize={35}
      width="100%"
    >
      {Row}
    </FixedSizeList>
  );
}

手动实现核心逻辑 ​

javascript
class VirtualList {
  constructor(container, items, itemHeight) {
    this.container = container;
    this.items = items;
    this.itemHeight = itemHeight;
    this.visibleCount = Math.ceil(container.clientHeight / itemHeight);
    this.startIndex = 0;
    
    this.container.addEventListener('scroll', () => this.handleScroll());
  }
  
  handleScroll() {
    this.startIndex = Math.floor(
      this.container.scrollTop / this.itemHeight
    );
    this.render();
  }
  
  render() {
    const fragment = document.createDocumentFragment();
    const endIndex = Math.min(
      this.startIndex + this.visibleCount,
      this.items.length
    );
    
    for (let i = this.startIndex; i < endIndex; i++) {
      const item = document.createElement('div');
      item.style.height = this.itemHeight + 'px';
      item.textContent = this.items[i];
      fragment.appendChild(item);
    }
    
    this.container.innerHTML = '';
    this.container.appendChild(fragment);
  }
}

4. JS 执行优化 ​

4.1 主线程减负 ​

核心目标：长任务拆分，利用 requestIdleCallback 在浏览器空闲时执行非关键逻辑。

长任务拆分 ​

javascript
// ❌ 不好：阻塞主线程
function heavyTask() {
  for (let i = 0; i < 100000000; i++) {
    // 大量计算
  }
}

// ✅ 好：拆分任务
function heavyTask() {
  const TASK_SIZE = 1000;
  let index = 0;
  
  function executeChunk() {
    const endIndex = Math.min(index + TASK_SIZE, 100000000);
    
    for (let i = index; i < endIndex; i++) {
      // 计算
    }
    
    index = endIndex;
    
    if (index < 100000000) {
      // 让出主线程
      setTimeout(executeChunk, 0);
    }
  }
  
  executeChunk();
}

requestIdleCallback ​

javascript
// 在浏览器空闲时执行非关键任务
requestIdleCallback((deadline) => {
  // deadline.timeRemaining() 返回剩余空闲时间
  while (deadline.timeRemaining() > 1 && hasMoreWork()) {
    doWork();
  }
  
  if (hasMoreWork()) {
    requestIdleCallback(myCallback);
  }
});

// 可设置超时时间确保执行
requestIdleCallback(myCallback, { timeout: 2000 });

4.2 脚本加载策略 ​

核心目标：非核心 JS 增加 defer/async，避免阻塞渲染（例如：统计、广告）。

html
<!-- 默认：阻塞渲染 -->
<script src="critical.js"></script>

<!-- defer：异步加载，DOM 解析完后执行（保持顺序） -->
<script src="app.js" defer></script>

<!-- async：异步加载，下载完立即执行（不保持顺序） -->
<script src="analytics.js" async></script>

<!-- 关键脚本内联 -->
<script>
  // 关键代码直接写在 HTML 中
</script>

<!-- 非关键脚本延迟加载 -->
<script defer src="non-critical.js"></script>

动态加载脚本 ​

javascript
// 按需加载脚本
function loadScriptDynamically(src) {
  return new Promise((resolve, reject) => {
    const script = document.createElement('script');
    script.src = src;
    script.onload = resolve;
    script.onerror = reject;
    document.body.appendChild(script);
  });
}

// 用户交互时加载
document.addEventListener('click', () => {
  loadScriptDynamically('/analytics.js');
});

面试高频问题 ​

1. 如何衡量页面加载速度？ ​

答：通过 Core Web Vitals（LCP、FID、CLS）以及 FCP、TTI、TTFB 等指标。

2. 首屏加载时间 2s 如何优化到 1s？ ​

答：

  • 代码分割和路由懒加载
  • 图片优化和 CDN 加速
  • 关键 CSS 内联、脚本延迟加载
  • 使用 SSR 或预渲染
  • 长任务拆分，减少 JS 执行时间

3. Tree-shaking 和代码分割的区别？ ​

答：

  • Tree-shaking：删除未使用的代码
  • 代码分割：按需加载代码块

4. 如何避免首屏白屏？ ​

答：使用骨架屏、关键样式内联、SSR 或服务端模板。

5. 虚拟滚动实现原理？ ​

答：根据滚动位置计算可见范围，只渲染可视区域的 DOM，其余部分用高度撑开。
